<script>
  const taskInput = document.getElementById('taskInput');
  const subjectInput = document.getElementById('subjectInput');
  const timeInput = document.getElementById('timeInput');
  const addBtn = document.getElementById('addBtn');
  const todoList = document.getElementById('todoList');
  const emptyState = document.getElementById('emptyState');
  const alarmSound = document.getElementById('alarmSound');

  let tasks = [];
  let idCounter = 0;

  if ('Notification' in window && Notification.permission === 'default') {
    Notification.requestPermission();
  }

  addBtn.addEventListener('click', addTask);
  taskInput.addEventListener('keypress', e => { if (e.key === 'Enter') addTask(); });

  function addTask() {
    const name = taskInput.value.trim();
    const subject = subjectInput.value.trim();
    const time = timeInput.value;

    if (!name) { taskInput.focus(); return; }
    if (!time) { timeInput.focus(); return; }

    const alarmTime = new Date(time).getTime();
    if (alarmTime < Date.now()) { alert('Please choose a time in the future'); return; }

    const task = {
      id: idCounter++,
      name, subject,
      time: alarmTime,
      done: false,
      ringing: false,
      timeoutId: null
    };

    tasks.push(task);
    scheduleAlarm(task);
    renderTasks();

    taskInput.value = '';
    subjectInput.value = '';
    timeInput.value = '';
    taskInput.focus();
  }

  function scheduleAlarm(task) {
    const delay = task.time - Date.now();
    task.timeoutId = setTimeout(() => triggerAlarm(task.id), delay);
  }

  function triggerAlarm(id) {
    const task = tasks.find(t => t.id === id);
    if (!task || task.done) return;

    task.ringing = true;
    renderTasks();

    alarmSound.play().catch(() => {});

    if ('Notification' in window && Notification.permission === 'granted') {
      new Notification(task.name, { body: task.subject || 'Alarm time' });
    } else {
      document.title = '● ' + task.name;
    }
  }

  function stopAlarm(id) {
    const task = tasks.find(t => t.id === id);
    if (!task) return;
    task.ringing = false;
    alarmSound.pause();
    alarmSound.currentTime = 0;
    document.title = "Day Planner — Alarms";
    renderTasks();
  }

  function toggleDone(id) {
    const task = tasks.find(t => t.id === id);
    if (!task) return;
    task.done = !task.done;
    if (task.done) {
      clearTimeout(task.timeoutId);
      task.ringing = false;
      alarmSound.pause();
    }
    renderTasks();
  }

  function deleteTask(id) {
    const task = tasks.find(t => t.id === id);
    if (task) clearTimeout(task.timeoutId);
    tasks = tasks.filter(t => t.id !== id);
    renderTasks();
  }

  function formatTime(ts) {
    return new Date(ts).toLocaleString(undefined, {
      month: 'short', day: 'numeric', hour: 'numeric', minute: '2-digit'
    });
  }

  function formatCountdown(ts) {
    const diff = ts - Date.now();
    if (diff <= 0) return 'now';
    const s = Math.floor(diff / 1000);
    const d = Math.floor(s / 86400);
    const h = Math.floor((s % 86400) / 3600);
    const m = Math.floor((s % 3600) / 60);
    const sec = s % 60;

    let parts = [];
    if (d > 0) parts.push(d + 'd');
    if (h > 0 || d > 0) parts.push(h + 'h');
    if (m > 0 || h > 0 || d > 0) parts.push(m + 'm');
    if (d === 0 && h === 0) parts.push(sec + 's');

    return 'in ' + parts.join(' ');
  }

  function renderTasks() {
    todoList.innerHTML = '';
    const sorted = tasks.slice().sort((a, b) => a.time - b.time);

    emptyState.style.display = sorted.length ? 'none' : 'block';

    sorted.forEach(task => {
      const entry = document.createElement('div');
      entry.className = 'entry' + (task.done ? ' done' : '') + (task.ringing ? ' ringing' : '');

      const rail = document.createElement('div');
      rail.className = 'rail';
      const dot = document.createElement('div');
      dot.className = 'dot';
      rail.appendChild(dot);

      const body = document.createElement('div');
      body.className = 'entry-body';

      const top = document.createElement('div');
      top.className = 'entry-top';

      const nameEl = document.createElement('span');
      nameEl.className = 'task-name';
      nameEl.textContent = task.name;
      top.appendChild(nameEl);

      if (task.subject) {
        const subjEl = document.createElement('span');
        subjEl.className = 'task-subject';
        subjEl.textContent = task.subject;
        top.appendChild(subjEl);
      }

      const meta = document.createElement('div');
      meta.className = 'meta-line';

      const timeEl = document.createElement('span');
      timeEl.textContent = formatTime(task.time);
      meta.appendChild(timeEl);

      if (!task.done) {
        const cd = document.createElement('span');
        cd.className = 'countdown';
        cd.textContent = task.ringing ? 'ringing now' : formatCountdown(task.time);
        meta.appendChild(cd);
      }

      body.appendChild(top);
      body.appendChild(meta);

      const actions = document.createElement('div');
      actions.className = 'entry-actions';

      if (task.ringing) {
        const stopBtn = document.createElement('button');
        stopBtn.className = 'stop-pill';
        stopBtn.textContent = 'Stop';
        stopBtn.onclick = () => stopAlarm(task.id);
        actions.appendChild(stopBtn);
      }

      const doneBtn = document.createElement('button');
      doneBtn.className = 'icon-btn';
      doneBtn.textContent = task.done ? '↺' : '✓';
      doneBtn.title = task.done ? 'Mark not done' : 'Mark done';
      doneBtn.onclick = () => toggleDone(task.id);
      actions.appendChild(doneBtn);

      const delBtn = document.createElement('button');
      delBtn.className = 'icon-btn';
      delBtn.textContent = '×';
      delBtn.title = 'Remove';
      delBtn.onclick = () => deleteTask(task.id);
      actions.appendChild(delBtn);

      entry.appendChild(rail);
      entry.appendChild(body);
      entry.appendChild(actions);
      todoList.appendChild(entry);
    });
  }

  setInterval(() => { if (tasks.length) renderTasks(); }, 1000);

  renderTasks();
</script>
