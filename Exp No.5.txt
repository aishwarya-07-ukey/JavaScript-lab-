<script>
  // ---------- Object handling: cart items are stored as objects in an array ----------
  let cart = [];

  // Closure: createLogger() returns a function that remembers `count` and
  // `entries` between calls, even though createLogger() itself has finished running.
  function createLogger() {
    let count = 0;
    let entries = [];
    return function (message) {
      count++;
      entries = [...entries, `Check #${count}: ${message}`];
      return entries;
    };
  }
  const logEntry = createLogger();

  function getDiscountRate(subtotal) {
    if (subtotal >= 5000) return 0.20;
    if (subtotal >= 2000) return 0.10;
    return 0;
  }

  function calculateCart() {
    // filter() removes any invalid/zero-quantity items defensively
    // map() converts each item object into its line subtotal
    // reduce() aggregates the subtotals into a single cart subtotal
    const subtotal = cart
      .filter(item => item.qty > 0)
      .map(item => item.price * item.qty)
      .reduce((total, amount) => total + amount, 0);

    const discountRate = getDiscountRate(subtotal);
    const discountAmount = subtotal * discountRate;
    const total = subtotal - discountAmount;

    return { subtotal, discountRate, discountAmount, total };
  }

  function render() {
    const tbody = document.getElementById('itemsBody');
    tbody.innerHTML = '';

    if (cart.length === 0) {
      tbody.innerHTML = '<tr class="empty-row"><td colspan="5">Cart is empty — add an item above</td></tr>';
    } else {
      cart.forEach((item, index) => {
        // Object destructuring to pull fields out of each cart item object
        const { name, qty, price } = item;
        const row = document.createElement('tr');
        row.innerHTML = `
          <td>${name}</td>
          <td>${qty}</td>
          <td>₹${price.toFixed(2)}</td>
          <td>₹${(price * qty).toFixed(2)}</td>
          <td class="remove" data-index="${index}">✕</td>`;
        tbody.appendChild(row);
      });
    }

    const { subtotal, discountRate, discountAmount, total } = calculateCart();
    document.getElementById('subtotalVal').textContent = `₹${subtotal.toFixed(2)}`;
    document.getElementById('discountVal').textContent = `-₹${discountAmount.toFixed(2)}`;
    document.getElementById('tierBadge').textContent = `${Math.round(discountRate * 100)}%`;
    document.getElementById('totalVal').textContent = `₹${total.toFixed(2)}`;

    Array.prototype.forEach.call(document.querySelectorAll('.remove'), function (el) {
      el.addEventListener('click', function () {
        var idx = Number(el.getAttribute('data-index'));
        cart = cart.filter(function (_, i) { return i !== idx; }); // filter() removes by index
        render();
      });
    });
  }

  document.getElementById('addForm').addEventListener('submit', function (e) {
    e.preventDefault();
    const errorMsg = document.getElementById('errorMsg');
    errorMsg.textContent = '';

    try {
      const name = document.getElementById('itemName').value.trim();
      const qty = Number(document.getElementById('itemQty').value);
      const price = Number(document.getElementById('itemPrice').value);

      if (!name) throw new Error('Item name cannot be empty');
      if (!qty || qty <= 0) throw new Error('Quantity must be greater than 0');
      if (price < 0 || Number.isNaN(price)) throw new Error('Price must be a valid, non-negative number');

      // New item object added to the cart array (spread used to keep future updates immutable)
      cart = [...cart, { name, qty, price }];

      const { subtotal, discountRate, total } = calculateCart();
      const entries = logEntry(
        `Added "${name}" -> Subtotal ₹${subtotal.toFixed(2)} -> Tier ${Math.round(discountRate * 100)}% -> Total ₹${total.toFixed(2)}`
      );

      this.reset();
      render();
      renderLogList(entries);
    } catch (err) {
      errorMsg.textContent = err.message; // caught by try-catch, shown to the user
    } finally {
      document.getElementById('itemName').focus();
    }
  });

  function renderLogList(entries) {
    document.getElementById('logList').innerHTML =
      entries.map(e => `<div class="log-entry">${e}</div>`).join('');
  }

  render();
</script>
