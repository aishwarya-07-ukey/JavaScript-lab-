<script>
  const fullNameEl = document.getElementById('fullName');
  const phoneEl = document.getElementById('phone');
  const emailEl = document.getElementById('email');
  const summaryEl = document.getElementById('summary');

  const phoneError = document.getElementById('phoneError');
  const emailError = document.getElementById('emailError');

  const charCountEl = document.getElementById('charCount');
  const wordCountEl = document.getElementById('wordCount');
  const sentenceCountEl = document.getElementById('sentenceCount');

  const phoneRegex = /^[0-9]{10}$/;
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

  function updateStats(){
    const text = summaryEl.value;
    const trimmed = text.trim();

    const charCount = text.length;
    const wordCount = trimmed.length === 0 ? 0 : trimmed.split(/\s+/).length;
    const sentenceMatches = trimmed.match(/[^.!?]+[.!?]+/g);
    const sentenceCount = sentenceMatches ? sentenceMatches.length : (trimmed.length ? 1 : 0);

    charCountEl.textContent = charCount;
    wordCountEl.textContent = wordCount;
    sentenceCountEl.textContent = sentenceCount;
  }

  summaryEl.addEventListener('input', updateStats);

  document.getElementById('btnUpper').addEventListener('click', () => {
    summaryEl.value = summaryEl.value.toUpperCase();
    updateStats();
  });

  document.getElementById('btnLower').addEventListener('click', () => {
    summaryEl.value = summaryEl.value.toLowerCase();
    updateStats();
  });

  document.getElementById('btnFixSpaces').addEventListener('click', () => {
    summaryEl.value = summaryEl.value.replace(/ {2,}/g, ' ');
    updateStats();
  });

  function validatePhone(){
    const val = phoneEl.value.trim();
    const valid = phoneRegex.test(val);
    phoneEl.classList.toggle('invalid', !valid && val.length > 0);
    phoneError.classList.toggle('show', !valid && val.length > 0);
    return valid;
  }

  function validateEmail(){
    const val = emailEl.value.trim();
    const valid = emailRegex.test(val);
    emailEl.classList.toggle('invalid', !valid && val.length > 0);
    emailError.classList.toggle('show', !valid && val.length > 0);
    return valid;
  }

  phoneEl.addEventListener('input', validatePhone);
  emailEl.addEventListener('input', validateEmail);

  document.getElementById('submitBtn').addEventListener('click', () => {
    const nameVal = fullNameEl.value.trim();
    const phoneValid = validatePhone();
    const emailValid = validateEmail();
    const phoneVal = phoneEl.value.trim();
    const emailVal = emailEl.value.trim();

    if (!nameVal || !phoneValid || !emailVal || !emailValid){
      document.getElementById('resultCard').classList.remove('show');
      if (!phoneVal) { phoneEl.classList.add('invalid'); phoneError.classList.add('show'); }
      if (!emailVal) { emailEl.classList.add('invalid'); emailError.classList.add('show'); }
      return;
    }

    const formattedName = nameVal
      .split(/\s+/)
      .map(w => w.toUpperCase())
      .join(' ');

    const summaryTrimmed = summaryEl.value.trim().replace(/ {2,}/g, ' ');
    const words = summaryTrimmed.length ? summaryTrimmed.split(/\s+/).length : 0;
    const chars = summaryTrimmed.length;
    const sentMatches = summaryTrimmed.match(/[^.!?]+[.!?]+/g);
    const sentences = sentMatches ? sentMatches.length : (summaryTrimmed.length ? 1 : 0);

    document.getElementById('outName').textContent = formattedName;
    document.getElementById('outPhone').textContent = phoneVal;
    document.getElementById('outEmail').textContent = emailVal;
    document.getElementById('outStats').textContent = `${words} words | ${chars} characters | ${sentences} sentences`;
    document.getElementById('outBio').textContent = summaryTrimmed || '(No summary provided.)';

    document.getElementById('resultCard').classList.add('show');
    fullNameEl.value = formattedName;
  });

  updateStats();
</script>
